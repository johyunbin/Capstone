## 🔧 빌드 환경

```bash

python3 _internal/scripts/_build_docx_v0.py
# (출력: submission/_drafts/속도는벡터_중간보고서_<타임코드>.docx)

# docx → PDF (Microsoft Word 자동화)
osascript << 'EOF'
tell application "Microsoft Word"
  activate
  open "<docx 경로>"
  set theDoc to active document
  save as theDoc file name "<pdf 경로>" file format format PDF
  close theDoc saving no
  quit
end tell
EOF

# md → PDF
python3 _internal/scripts/md2pdf.py <md 경로>
```
