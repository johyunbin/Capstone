def stratified_reservoir(stream, K=20, N=385):
    R = [[] for _ in range(K)]  # K reservoir
    n = [0] * K                   # per-cluster count
    cap = [N//K] * K              # Equal default

    for tuple_t in stream:
        j = cluster_assign(tuple_t, K)  # BIRCH CF
        n[j] += 1
        if len(R[j]) < cap[j]:
            R[j].append(tuple_t)
        else:
            k = random.randint(0, n[j]-1)
            if k < cap[j]:
                R[j][k] = tuple_t  # Vitter Algo R
    return R
